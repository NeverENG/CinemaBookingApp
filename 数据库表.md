# 用户表
CREATE USER TABLE(
  id uuid PRIVATE KEY NOT NULL,
  name TEXT NOT NULL,
  Role TEXT NOT NULL
  Check (role in ('user','admin','supplier_admin'))
  email TEXT NOT NULL,
  password TEXT NOT NULL,
  avatar INT NOT NULL, # 自选 6 个头像选一个
  level TEXT NOT NULL Default 'Basic'
  Check (level in ('Basic','Premium','VIP')),
  points INT NOT NULL Default 0,
  create_time TIMESTAMP NOT NULL,
  update_time TIMESTAMP NOT NULL,
)
